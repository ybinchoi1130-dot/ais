[판다스 설치]

###########################################
# 아나콘다 파워셀
###########################################

# 가상환경 목록 확인
conda env list

# 가상환경 활성화
conda activate ais

# 가상환경 비활성화
conda deactivate ais

###########################################
# 파이썬 패키지 설치 도구
###########################################

# 설치된 패키지 목록 확인
pip list

# 패키지 설치
pip install pandas

# 패키지 확인
pip show pandas

# 버전
# numpy-2.5.2
# pandas-3.0.5

# 패키지 설치 제거
pip uninstall pandas
